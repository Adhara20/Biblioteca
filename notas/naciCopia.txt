	<!-- JS para que jale el Menú desplegable -->
	<script>
	document.addEventListener("click", (e) => {
    document.querySelectorAll(".btn-kebab").forEach(btn => {
        const menu = btn.nextElementSibling;

        // clic en el botón
        if (btn.contains(e.target)) {
            // abrir/cerrar
            menu.classList.toggle("hidden");

            // ---- Ajustar dirección del menú ----
            const rect = menu.getBoundingClientRect();
            const spaceBelow = window.innerHeight - rect.bottom;
            const spaceAbove = rect.top;

            // Si no cabe abajo, abrir hacia arriba
            if (spaceBelow < 100 && spaceAbove > 100) {
                menu.classList.remove("top-full", "mt-1");
                menu.classList.add("bottom-full", "mb-1");
            } else {
                // posición normal
                menu.classList.remove("bottom-full", "mb-1");
                menu.classList.add("top-full", "mt-1");
            }
        } 
        else {
            menu.classList.add("hidden");
        }
		});
	});
	</script>
